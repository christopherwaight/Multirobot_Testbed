\section*{Local center/direction estimation from a single sample \texorpdfstring{$(v,J)$}{(v,J)}}

\subsection*{Setup and notation}
Let \(v:\Omega\subset\mathbb{R}^2\to\mathbb{R}^2\) be \(C^2\) on a punctured neighborhood of an unknown point \(c\) (the ``center''). 
We observe the field at a single point \(x\in\Omega\), with
\[
v := v(x)\in\mathbb{R}^2,\qquad J := Dv(x)\in\mathbb{R}^{2\times 2}.
\]
Define the scalar \emph{speed} and the unit flow direction
\[
V:=\|v\|,\qquad t:=\frac{v}{\|v\|}\quad(\text{if }V>0).
\]
We will use the following \emph{local quadratic probes}, computable from \((v,J)\) at \(x\):
\begin{align}
g &:= J^\top v \;=\; \nabla\!\left(\tfrac12\|v\|^2\right) \in\mathbb{R}^2, \label{eq:g-def}\\[2pt]
s &:= J\,v \;=\; (v\!\cdot\!\nabla)\,v \in\mathbb{R}^2, \label{eq:s-def}\\[2pt]
s_\parallel &:= (t^\top s)\,t,\qquad s_\perp := s - s_\parallel, \label{eq:s-split}
\end{align}
as well as the familiar invariants
\[
\operatorname{div} v = \mathrm{tr}\,J,\qquad 
\omega := (\nabla\times v)_z = \partial_x v_y - \partial_y v_x.
\]
Intuitively: \(g\) points toward increasing speed, and \(s_\perp\) captures the streamline curvature (normal acceleration).

\subsection*{Model class near the center}
Many fields of interest near an isolated center \(c\) can be written (in polar basis \(\{\hat e_r,\hat e_\theta\}\) about \(c\)) as
\begin{equation}
v(x) \;=\; \alpha(r)\,\hat e_r \;+\; \beta(r)\,\hat e_\theta,\qquad r:=\|x-c\|, \label{eq:polar-form}
\end{equation}
with speed \(V(r)=\sqrt{\alpha(r)^2+\beta(r)^2}\). Your six families correspond to
\[
V(r)\in\{ \kappa r^{-1},\; \kappa,\; \kappa r \}\quad\text{for sinks/sources (\(\alpha\) dominant) and vortices (\(\beta\) dominant).}
\]

\subsection*{Main estimators (direction and distance to the center)}
\paragraph{(A) Direction to the center.}
\begin{equation}
\boxed{\;\hat u_{\text{rad}} \;:=\; -\,\frac{g}{\|g\|}\quad\text{if }\|g\|>0\;}\label{eq:dir-grad}
\end{equation}
This points along \(-\nabla(\tfrac12 V^2)\), i.e., inward (toward increasing speed) whenever \(V\) primarily depends on \(r\).\\
If \(\|g\|\approx 0\) but the flow is curved (e.g., vortex with nearly constant \(V\)),
\begin{equation}
\boxed{\;\hat u_{\text{curv}} \;:=\; -\,\frac{s_\perp}{\|s_\perp\|}\quad\text{if }\|s_\perp\|>0\;}\label{eq:dir-curv}
\end{equation}
This points toward the osculating-circle center of the streamline.

\paragraph{(B) Distance to the center.} Use the signal that is active:
\begin{align}
\text{(radial-speed cue)}\qquad 
&\boxed{\; r \;=\; \frac{V^2}{\|g\|}\;}\quad 
\text{exact for }V(r)\propto r \text{ and } V(r)\propto 1/r. 
\label{eq:r-from-g}\\[6pt]
\text{(curvature cue)}\qquad
&\boxed{\; r \;=\; \frac{V^2}{\|s_\perp\|}\;}\quad 
\text{for circular/vortical motion (any }V(r)\text{).}
\label{eq:r-from-sperp}
\end{align}
Special constant-magnitude shortcuts:
\begin{align}
&\text{radial constant-speed sink/source:} && 
\boxed{\; r \;=\; \frac{V}{|\operatorname{div} v|}\;},\qquad \text{direction along }\pm t, 
\label{eq:const-radial}\\[2pt]
&\text{tangential constant-speed vortex:} &&
\boxed{\; r \;=\; \frac{V}{|\omega|}\;}\quad\text{(and also }r=\tfrac{V^2}{\|s_\perp\|}\text{).}
\label{eq:const-vortex}
\end{align}

\subsection*{Why these formulas work (proof sketches)}
Write \(V^2(r)=F(r)\). Then \(f:=\tfrac12 V^2=\tfrac12 F(r)\) has
\[
\nabla f = \frac{F'(r)}{2}\,\hat e_r \quad\Rightarrow\quad g=J^\top v=\nabla f\ \text{ is radial.}
\]
Hence \(\hat u_{\text{rad}}=-g/\|g\|\) points toward the center when \(F\) increases inward (typical near sinks/vortices). 
If \(V(r)=\kappa r^p\) with \(p\in\{-1,0,1\}\), then
\[
\|g\| \;=\; \left\|\nabla \tfrac12 V^2\right\| \;=\; \left\|\nabla \tfrac12 \kappa^2 r^{2p}\right\| \;=\; |\kappa|^2\,|p|\, r^{2p-1},
\]
and since \(V^2=\kappa^2 r^{2p}\),
\[
\frac{V^2}{\|g\|} \;=\; \frac{\kappa^2 r^{2p}}{|\kappa|^2 |p| r^{2p-1}} \;=\; \frac{r}{|p|}.
\]
Thus \(r=|p|\,V^2/\|g\|\). For the two nontrivial cases \(p=\pm1\) we obtain \eqref{eq:r-from-g} exactly; when \(p=0\) (constant \(V\)), \(g=0\) and the gradient cue is silent (as desired).

For vortical motion, the normal (inward) component of the convective derivative is the centripetal acceleration:
\[
\|s_\perp\| \;=\; \frac{V^2}{R}\quad\Rightarrow\quad R \;=\; \frac{V^2}{\|s_\perp\|},
\]
where \(R\) is the osculating radius of the streamline. For circular motion about \(c\), \(R=r\), giving \eqref{eq:r-from-sperp}. 
Equation \eqref{eq:const-vortex} follows from \(\omega=2\Omega\) for rigid rotation \(V=\Omega r\) (so \(r=2V/|\omega|\)) and \(\omega=V/r\) for constant-speed tangential flow (so \(r=V/|\omega|\)). 
For a radial constant-speed sink, \(\operatorname{div}(\pm V\,\hat e_r) = \pm V/r\), hence \eqref{eq:const-radial}.

\subsection*{Decision procedure (single-sample, robust)}
Given \((v,J)\) at \(x\) with \(V=\|v\|>0\):
\begin{enumerate}
\item Compute \(g=J^\top v\) and \(s_\perp = Jv - (t^\top Jv)\,t\) with \(t=v/V\).
\item If \(\|g\|\) is not tiny, set direction \(\hat u=-g/\|g\|\) and distance \(r=V^2/\|g\|\) (covers \(V\sim r\) and \(V\sim 1/r\) for sinks and vortices).
\item Else if \(\|s_\perp\|\) is not tiny, set direction \(\hat u=-s_\perp/\|s_\perp\|\) and distance \(r=V^2/\|s_\perp\|\) (vortices, including constant-speed).
\item Else (both cues tiny): the center is not locally observable from a single sample (e.g., nearly uniform straight flow). 
   If \(|\operatorname{div} v|\) is appreciable and \(t\) is approximately radial, you may use \(r=V/|\operatorname{div} v|\) with direction \(\pm t\) per \eqref{eq:const-radial}.
\end{enumerate}

\subsection*{Verification on the six canonical families}
Let \(r=\|x-c\|\).
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{lllc}
\hline
Type & Representative field & Active cue(s) & Recovered \(r\) \\
\hline
Sink/source, \(V\sim 1/r\) & \(v=\pm \kappa\,\hat e_r/r\) & \(g\) & \(r=V^2/\|g\|\) (exact) \\ 
Sink/source, \(V=\text{const}\) & \(v=\pm \kappa\,\hat e_r\) & \(\operatorname{div} v\) & \(r=V/|\operatorname{div} v|\) (exact) \\
Sink/source, \(V\sim r\) & \(v=\pm \kappa\,r\,\hat e_r\) & \(g\) & \(r=V^2/\|g\|\) (exact) \\
Vortex, \(V\sim 1/r\) & \(v=\kappa\,\hat e_\theta/r\) & \(g,\ s_\perp\) & \(r=V^2/\|g\|=V^2/\|s_\perp\|\) (exact) \\
Vortex, \(V=\text{const}\) & \(v=\kappa\,\hat e_\theta\) & \(s_\perp\) & \(r=V^2/\|s_\perp\|=V/|\omega|\) (exact) \\
Vortex, \(V\sim r\) & \(v=\kappa\,r\,\hat e_\theta\) & \(g,\ s_\perp\) & \(r=V^2/\|g\|=2V/|\omega|\) (exact) \\
\hline
\end{tabular}
\end{center}

\subsection*{Practical notes and caveats}
\begin{itemize}
\item The ``direction'' rules \eqref{eq:dir-grad} and \eqref{eq:dir-curv} do not assume ``sink vs vortex''; they rely on the radiality of \(\nabla(\tfrac12 V^2)\) and the inward normal acceleration of curved streamlines.
\item In a \emph{sinking vortex} (both radial and tangential parts present), both cues are typically nonzero and yield consistent radii \(r\).
\item When nearby boundaries, multiple centers, or strong anisotropy break the \(r\)-only dependence of \(V\), a single local sample can be insufficient; take a second sample and re-evaluate.
\item Numerics: If \(\|g\|\) or \(\|s_\perp\|\) is tiny relative to \(V\|J\|\), treat the corresponding estimate as unreliable.
\end{itemize}
