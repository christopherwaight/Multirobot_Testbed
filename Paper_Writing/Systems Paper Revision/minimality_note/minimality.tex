\subsection*{Minimum Robot Requirements}

\noindent\textbf{Proposition 1 (Minimality):} \textit{For an affine field $\mathbf{v}(\mathbf{p}) = \mathbf{J}(\mathbf{p}-\mathbf{p}^*)$ with $\det(\mathbf{J}) \neq 0$, locating $\mathbf{p}^*$ of every such field from instantaneous measurements requires at least three non-collinear robots, and three suffice.}

\smallskip
\noindent\textit{Proof:} \textit{Necessity.} Any $n \le 2$ (or collinear) robots lie on a line $L = \{\mathbf{p}_0 + s\mathbf{d}\}$ and measure $\mathbf{m}_0 + s_i\mathbf{w}$, where $\mathbf{m}_0 = \mathbf{J}(\mathbf{p}_0 - \mathbf{p}^*)$ and $\mathbf{w} = \mathbf{J}\mathbf{d}$. If $\mathbf{p}^* \notin L$, then for any $\mathbf{q} \notin L$ the field $\mathbf{J}_q(\mathbf{p} - \mathbf{q})$ with $\mathbf{J}_q = [\,\mathbf{m}_0\ \ \mathbf{w}\,][\,\mathbf{p}_0 - \mathbf{q}\ \ \mathbf{d}\,]^{-1}$ is nonsingular and reproduces every measurement. Its determinant changes sign across $L$, so neither $\mathbf{p}^*$ nor its type is determined. \textit{Sufficiency.} $\mathbf{A}$ in (\ref{eq:4}) loses column rank only when a nonzero affine function vanishes at every robot, that is, for collinear robots. Three non-collinear robots therefore give $\hat{\mathbf{J}} = \mathbf{J}$ and, via (\ref{eq:9}), $\hat{\mathbf{p}}^* = \mathbf{p}^*$, as does least squares for $n > 3$. \hfill\IEEEQED
